              async function initializeFirebaseIfReady() {
                  if (window.db) {
                      updateFirebaseConnectionBadges(true);
                      return true;
                  }
                  if (isFirebaseInitInProgress) return false;
                  isFirebaseInitInProgress = true;
                  updateFirebaseConnectionBadges(false);
                  
                  try {
                      console.log("[Firebase Applet] Starting dynamic import loader for modern modular SDK...");
                      const { initializeApp } = await import("https://www.gstatic.com/firebasejs/10.12.0/firebase-app.js");
                      const { 
                          initializeFirestore, 
                          collection, 
                          doc, 
                          setDoc, 
                          deleteDoc,
                          updateDoc, 
                          getDocs, 
                          getDoc,
                          onSnapshot,
                          query,
                          orderBy,
                          limit,
                          runTransaction
                      } = await import("https://www.gstatic.com/firebasejs/10.12.0/firebase-firestore.js");

                      const {
                          getAuth,
                          GoogleAuthProvider,
                          signInWithPopup,
                          signOut
                      } = await import("https://www.gstatic.com/firebasejs/10.12.0/firebase-auth.js");

                      let app = initializeApp(firebaseConfig);
                      const rawDb = initializeFirestore(app, {
                          experimentalForceLongPolling: true,
                          useFetchStreams: false
                      }, "ai-studio-849e4512-323e-4cb4-b68a-951d9ae13878");
                      console.log("[Firebase Applet] Connected directly to custom named database via modular SDK with long polling enabled!");

                      // Initialize Auth globals
                      window.firebaseAuth = getAuth(app);
                      window.GoogleAuthProvider = GoogleAuthProvider;
                      window.signInWithPopup = signInWithPopup;
                      window.signOut = signOut;

                      // Expose fully compliant compat wrapper on window.db supporting queries/get constraints
                      window.db = {
                          runTransaction: (updateFunction) => {
                              return runTransaction(rawDb, (transaction) => {
                                  const compatTransaction = {
                                      get: async (docRefCompat) => {
                                          const snapshot = await transaction.get(docRefCompat._rawDocRef);
                                          return {
                                              exists: snapshot.exists(),
                                              data: () => snapshot.data(),
                                              id: snapshot.id
                                          };
                                      },
                                      set: (docRefCompat, data) => {
                                          transaction.set(docRefCompat._rawDocRef, data);
                                          return compatTransaction;
                                      },
                                      update: (docRefCompat, data) => {
                                          transaction.update(docRefCompat._rawDocRef, data);
                                          return compatTransaction;
                                      },
                                      delete: (docRefCompat) => {
                                          transaction.delete(docRefCompat._rawDocRef);
                                          return compatTransaction;
                                      }
                                  };
                                  return updateFunction(compatTransaction);
                              });
                          },
                          collection: (collectionName) => {
                              const colRef = collection(rawDb, collectionName);
                              
                              const makeQuery = (constraints) => {
                                  return {
                                      doc: (docId) => {
                                          const docRef = doc(rawDb, collectionName, String(docId));
                                          return {
                                              _rawDocRef: docRef,
                                              set: (data) => setDoc(docRef, data),
                                              update: (data) => updateDoc(docRef, data),
                                              delete: () => deleteDoc(docRef),
                                              get: async () => {
                                                  const snapshot = await getDoc(docRef);
                                                  return {
                                                      exists: snapshot.exists(),
                                                      data: () => snapshot.data(),
                                                      id: snapshot.id
                                                  };
                                              },
                                              onSnapshot: (onNext, onError) => {
                                                  return onSnapshot(docRef, (snapshot) => {
                                                      onNext({
                                                          exists: snapshot.exists(),
                                                          data: () => snapshot.data(),
                                                          id: snapshot.id
                                                      });
                                                  }, onError || (err => console.error("Doc onSnapshot error:", err)));
                                              }
                                          };
                                      },
                                      orderBy: (field, direction = "asc") => {
                                          return makeQuery([...constraints, orderBy(field, direction)]);
                                      },
                                      limit: (num) => {
                                          return makeQuery([...constraints, limit(num)]);
                                      },
                                      onSnapshot: (onNext, onError) => {
                                          const q = constraints.length > 0 ? query(colRef, ...constraints) : colRef;
                                          return onSnapshot(q, (querySnapshot) => {
                                              const docs = [];
                                              querySnapshot.forEach((docSnap) => {
                                                  docs.push({
                                                      exists: docSnap.exists(),
                                                      data: () => docSnap.data(),
                                                      id: docSnap.id
                                                  });
                                              });
                                              onNext({
                                                  forEach: (callback) => docs.forEach(callback),
                                                  docs: docs,
                                                  empty: docs.length === 0
                                              });
                                          }, onError || (err => console.error("Col onSnapshot error:", err)));
                                      },
                                      get: async () => {
                                          const q = constraints.length > 0 ? query(colRef, ...constraints) : colRef;
                                          const querySnapshot = await getDocs(q);
                                          const docs = [];
                                          querySnapshot.forEach((docSnap) => {
                                              docs.push({
                                                  exists: docSnap.exists(),
                                                  data: () => docSnap.data(),
                                                  id: docSnap.id
                                              });
                                          });
                                          return {
                                              forEach: (callback) => docs.forEach(callback),
                                              docs: docs,
                                              empty: docs.length === 0
                                          };
                                      }
                                  };
                              };

                              return makeQuery([]);
                          }
                      };

                      updateFirebaseConnectionBadges(true);
                      setupFirebaseRealtimeListeners();
                      processPendingSyncs();
                      executeStartupSettlementCleanSweep();
                      
                      if (userProfile && userProfile.phone && userProfile.password) {
                          updateAppUsersList(userProfile);
                      }
                      isFirebaseInitInProgress = false;
                      return true;
                  } catch (e) {
                      console.error("Firebase modular SDK init failed inside index.html:", e);
                      logFirebaseError("Modular SDK Init Failed", e);
                      isFirebaseInitInProgress = false;
                      updateFirebaseConnectionBadges(false);
                      
                      firebaseInitAttempts++;
                      if (firebaseInitAttempts < 30) {
                          setTimeout(initializeFirebaseIfReady, 1000);
                      } else {
                          console.warn("Firebase SDK initialization totally exhausted.");
                      }
                  }
                  return false;
              }

              // Run on load
              initializeFirebaseIfReady();
